14. Implement the CNN

Now we create our Convolutional Neural Network. 


class CNN(nn.Module):

    def __init__(self):

        super(CNN, self).__init__()

        # First convolution layer
        self.conv1 = nn.Conv2d(
            in_channels=3,
            out_channels=32,
            kernel_size=3,
            padding=1
        )

        # Second convolution layer
        self.conv2 = nn.Conv2d(
            in_channels=32,
            out_channels=64,
            kernel_size=3,
            padding=1
        )

        # Max pooling
        self.pool = nn.MaxPool2d(
            kernel_size=2,
            stride=2
        )

        # Fully connected layer
        self.fc1 = nn.Linear(
            64 * 8 * 8,
            128
        )

        # Output layer
        self.fc2 = nn.Linear(
            128,
            10
        )


    def forward(self, x):

        # First convolution block
        x = self.conv1(x)

        x = torch.relu(x)

        x = self.pool(x)


        # Second convolution block
        x = self.conv2(x)

        x = torch.relu(x)

        x = self.pool(x)


        # Flatten
        x = x.view(
            x.size(0),
            -1
        )


        # Fully connected layer
        x = self.fc1(x)

        x = torch.relu(x)


        # Output layer
        x = self.fc2(x)

        return x

17. Create the Model


model = CNN().to(device)

print(model)

18. Define Loss Function


criterion = nn.CrossEntropyLoss()

We use CrossEntropyLoss because this is a multi-class classification problem.

The model produces 10 output scores, and the correct class is compared with those scores to calculate the loss.

19. Define Optimizer


optimizer = optim.Adam(
    model.parameters(),
    lr=LEARNING_RATE
)

The optimizer updates the CNN's weights based on the gradients calculated during backpropagation.

20. Train the CNN


for epoch in range(EPOCHS):

    model.train()

    running_loss = 0.0

    for images, labels in train_loader:

        # Move images and labels to device
        images = images.to(device)
        labels = labels.to(device)

        # Clear old gradients
        optimizer.zero_grad()

        # Forward pass
        outputs = model(images)

        # Calculate loss
        loss = criterion(
            outputs,
            labels
        )

        # Backpropagation
        loss.backward()

        # Update weights
        optimizer.step()

        running_loss += loss.item()


    average_loss = (
        running_loss /
        len(train_loader)
    )

    print(
        f"Epoch [{epoch + 1}/{EPOCHS}] "
        f"Loss: {average_loss:.4f}"
    )

25. Class-Wise Evaluation

We can also calculate accuracy separately for each class. 


class_correct = [0] * 10
class_total = [0] * 10

model.eval()

with torch.no_grad():

    for images, labels in test_loader:

        images = images.to(device)
        labels = labels.to(device)

        outputs = model(images)

        _, predicted = torch.max(
            outputs,
            1
        )

        for i in range(len(labels)):

            label = labels[i].item()

            class_total[label] += 1

            if predicted[i] == labels[i]:

                class_correct[label] += 1


print("Class-wise Accuracy:")

for i in range(10):

    class_accuracy = (
        100 *
        class_correct[i] /
        class_total[i]
    )

    print(
        f"{classes[i]:12s}: "
        f"{class_accuracy:.2f}%"
    )

            in_channels=3,
            out_channels=32,
            kernel_size=3,
            padding=1
        )

        self.conv2 = nn.Conv2d(
            in_channels=32,
            out_channels=64,
            kernel_size=3,
            padding=1
        )

        self.pool = nn.MaxPool2d(
            kernel_size=2,
            stride=2
        )

        self.fc1 = nn.Linear(
            64 * 8 * 8,
            128
        )

        self.fc2 = nn.Linear(
            128,
            10

    def forward(self, x):

        x = self.conv1(x)
        x = torch.relu(x)
        x = self.pool(x)

        x = self.conv2(x)
        x = torch.relu(x)
